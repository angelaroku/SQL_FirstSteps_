CREATE TABLE regiones (
	region_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	region_nombre VARCHAR2 (25) DEFAULT NULL
);

CREATE TABLE paises (
	pais_id CHAR (2) PRIMARY KEY,
	pais_nombre VARCHAR2 (40) DEFAULT NULL,
	region_id NUMBER NOT NULL,
	FOREIGN KEY (region_id) REFERENCES regiones (region_id) ON DELETE CASCADE
);

CREATE TABLE ubicaciones (
	ubicacion_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	direccion VARCHAR2 (40) DEFAULT NULL,
	codigoPostal VARCHAR2 (12) DEFAULT NULL,
	ciudad VARCHAR2 (30) NOT NULL,
	provincia VARCHAR2 (25) DEFAULT NULL,
	pais_id CHAR (2) NOT NULL,
	FOREIGN KEY (pais_id) REFERENCES paises (pais_id) ON DELETE CASCADE
);

CREATE TABLE empleos (
	empleo_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	empleo_title VARCHAR2 (35) NOT NULL,
	min_salario NUMBER (8, 2) DEFAULT NULL,
	max_salario NUMBER (8, 2) DEFAULT NULL
);

CREATE TABLE departamentos (
	dpto_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	dpto_nombre VARCHAR2 (30) NOT NULL,
	ubicacion_id NUMBER DEFAULT NULL,
	FOREIGN KEY (ubicacion_id) REFERENCES ubicaciones (ubicacion_id) ON DELETE CASCADE
);

CREATE TABLE empleados (
	empleado_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	nombre VARCHAR2 (20) DEFAULT NULL,
	apellido VARCHAR2 (25) NOT NULL,
	email VARCHAR2 (100) NOT NULL,
	telefono VARCHAR2 (20) DEFAULT NULL,
	fecha_contrato DATE NOT NULL,
	empleo_id NUMBER NOT NULL,
	salario NUMBER (8, 2) NOT NULL,
	manager_id NUMBER DEFAULT NULL,
	dpto_id NUMBER DEFAULT NULL,
	FOREIGN KEY (empleo_id) REFERENCES empleos (empleo_id) ON DELETE CASCADE,
	FOREIGN KEY (dpto_id) REFERENCES departamentos (dpto_id) ON DELETE CASCADE,
	FOREIGN KEY (manager_id) REFERENCES empleados (empleado_id)
);

CREATE TABLE dependientes (
	dependent_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	nombre VARCHAR2 (50) NOT NULL,
	apellido VARCHAR2 (50) NOT NULL,
	relacion VARCHAR2 (25) NOT NULL,
	empleado_id NUMBER NOT NULL,
	FOREIGN KEY (empleado_id) REFERENCES empleados (empleado_id) ON DELETE CASCADE
);

--__--__--__--__--__--__--__--__---__---1. Obtén los primeros 5 caracteres del nombre del país y ordenalos alfabéticamente. Usa SUBSTR(cadena, pos1, long)

----------SOLUCION-----------------

SELECT pais_nombre,
	SUBSTR(pais_nombre, 1, 5) AS tablaPunto1
FROM paises
ORDER BY tablaPunto1 ASC;

----------EXPLICACION--------------
/*SOBRE EL "SUBSTR"--> es la accion utilizada para sustrater, o escoger una parte de una cadena de caracteres
por ejemplo, si tengo un elemento que se refiera a un nombre podria coger solo la inicial, o solo las primeras letras, o solo las ultimas
Se nos pide usar ""SUBSTR(cadena, pos1, long)"" -> ACCION(elemento, posicionInicialAescoger, UltimaPosicionAescoger)
	- Refiriendonos a un elemento llamado NOMBRE_PERSONA(TipoDeDato: VARCHAR2) :
Para una inicial por ejemplo--> SUBSTR(NOMBRE_PERSONA, 1, 1)
Para las 3 primeras letras por ejemplo-->  SUBSTR(NOMBRE_PERSONA, 1, 3)   */

--Ahora Por partes, leyendo el enunciado
/* separare el enunciado en 3 "misiones" para explicar mejor el proceso:
1.- se nos pide obtener la info de la columna de los nombre de los paises.
2.- recortar la info de los elementos de esa columna, con los 5 primeras letras.
3.- ordenar la info de manera alfabetica. */

----Mision1.--
/*Para estu, necesitaremos encontrar la tabla donde esta la columna que buscamos
si haceos select de todas las tablas; veremos la info que buscamos en la tabla paises*/
--tabla paises

select * from paises;

--selecionamos la columna que nos interesa: pais_nombre

SELECT pais_nombre
FROM paises;

----Mision2.--
/*Para esta mision, necesitaremos crear una columna "ficticia"
que noos muestre los datos de los paises "modificados" (solo sus 5 primeras letras)*/
--Usaremos el SUBTR explicado antes, y para adjuntar la nueva info, indicaremos 
--que la nueva informacion nos la añada como: "NOMBRE DE NUEVA COLUMNA " (yo la llame: TablaPunto1)

SELECT pais_nombre,
	SUBSTR(pais_nombre, 1, 5) AS tablaPunto1
FROM paises;

----Mision3.--
/*Solo nos queda ordenar la info de manera alfabetica
Para esO usaremos un ORDER BY refiriendonos a la nueva columna
y como lo queremos de alfabeticamente ponemos: ASC*/

SELECT pais_nombre,
	SUBSTR(pais_nombre, 1, 5) AS tablaPunto1
FROM paises
ORDER BY tablaPunto1 ASC;

/*En este caso si nos referimos a PAIS_NOMBRE en nuestro ORDER BY; el resultado seria el mismo,
pero si tuvieramos paises con nombres muy similares el resultado puede cambiar minusculamente; 
NOTA ***Para estos ejemplos, que creamos "columnas ficticias" si dudamos 
entre si nombrar la original o la modificacion de la "ficticia"; siempre mejor quedarse con la "ficticia"(la nueva columna)*/

SELECT pais_nombre,
	SUBSTR(pais_nombre, 1, 5) AS tablaPunto1
FROM paises
ORDER BY pais_nombre ASC;
