--1) Obtén la media de los precios de venta de los dispositivos de color Verde, Azul o Blanco:

SELECT AVG(p.prventa) AS media_precio_venta
FROM producto p, pedido pe, color c
where p.codprod = pe.producto_codprod
and  pe.color_codcolor = c.codcolor
and  c.nombre IN ('Verde', 'Azul', 'Blanco');

-- MEDIA_PRECIO_VENTA
-- 730


2) -- Obtener el código de los productos que han sido pedidos por 
-- un cliente de Córdoba en una tienda de Bilbao:

SELECT DISTINCT pe.producto_codprod
FROM pedido pe, cliente c, ciudad ci, tienda t
WHERE  pe.cliente_dni = c.dni
AND c.codciudad = ci.codciudad
AND pe.tienda_codtienda = t.codtienda 
AND ci.nombre = 'Córdoba' 
AND t.codciudad = (Select codciudad 
			from CIUDAD 
			where nombre='Bilbao')
ORDER BY 1;

-- PRODUCTO_CODPROD
-- 5

-- 3) Obtener el código de los productos pedidos en una 
--tienda de distinta ciudad que el cliente que los compra:

SELECT DISTINCT pe.producto_codprod
FROM pedido pe, cliente c,  tienda t
WHERE pe.cliente_dni = c.dni AND
  pe.tienda_codtienda = t.codtienda AND
  c.codciudad != t.codciudad
ORDER BY 1;

-- 4) Obtener los fabricantes que hayan fabricado algún producto 
--encargado por un cliente de Madrid en una tienda que no sea de Madrid:

SELECT DISTINCT f.nombre
FROM fabricante f
JOIN producto p ON f.codfabr = p.fabricante_codfabr
JOIN pedido pe ON p.codprod = pe.producto_codprod
JOIN cliente c ON pe.cliente_dni = c.dni
JOIN tienda t ON pe.tienda_codtienda = t.codtienda
WHERE c.codciudad = (Select codciudad from CIUDAD where nombre='Madrid')
AND t.codciudad != (Select codciudad from CIUDAD where nombre='Madrid')
ORDER BY F.NOMBRE;


-- NOMBRE
-- LG
-- Samsung

-- 5) Mostrar el nombre de modelo completo (nombre fabricante, marca y modelo) de aquellos productos que han sido 
---pedidos por clientes cuyo nombre empieza por 'A' o 'B' o su primer apellido acaba por 'a':

SELECT f.nombre || ' ' || p.marca || ' ' || p.modelo AS modelo_completo
FROM producto p
JOIN fabricante f ON p.fabricante_codfabr = f.codfabr
JOIN pedido pe ON p.codprod = pe.producto_codprod
JOIN cliente c ON pe.cliente_dni = c.dni
WHERE c.nombre LIKE 'A%' OR c.nombre LIKE 'B%' OR c.apellido1 LIKE '%a';

-- MODELO_COMPLETO
-- Samsung Samsung Galaxy S22
-- LG LG OLED TV
-- Samsung Samsung Galaxy S22
-- LG LG OLED TV
-- Panasonic Panasonic Lumix
-- Samsung Samsung Galaxy S22

-- 6) Obtener el DNI de los clientes que hayan pedido un producto de color 'Blanco' en una tienda de Madrid:

FROM pedido pe
JOIN color c ON pe.color_codcolor = c.codcolor
JOIN tienda t ON pe.tienda_codtienda = t.codtienda
WHERE c.nombre = 'Blanco' AND t.codciudad = (Select codciudad from CIUDAD where nombre='Madrid')
ORDER BY 1;

-- 8) Obtener la marca y el modelo de los productos 
--vendidos por alguna tienda que esté en Barcelona, Valencia, Málaga o Sevilla:

SELECT DISTINCT p.marca, p.modelo
FROM producto p
JOIN pedido pe ON p.codprod = pe.producto_codprod
JOIN tienda t ON pe.tienda_codtienda = t.codtienda
WHERE t.codciudad IN (Select codciudad
			from CIUDAD 
			where nombre IN ( 'Barcelona', 'Valencia', 'Málaga', 'Sevilla'))
ORDER BY 1,2;

-- MARCA    MODELO
-- LG    OLED TV
-- Panasonic    Lumix
-- Samsung    Galaxy S22
-- Sony    PS5

-- 9) Obtener el nombre y los apellidos de los clientes 
--cuya ciudad empiece por 'S' o 'B' que hayan adquirido un producto de color Azul o Negro:

SELECT DISTINCT c.nombre, c.apellido1, c.apellido2
FROM cliente c
JOIN pedido pe ON c.dni = pe.cliente_dni
JOIN color co ON pe.color_codcolor = co.codcolor
WHERE (c.codciudad IN (Select codciudad from CIUDAD where nombre LIKE 'S%' OR NOMBRE LIKE 'B%') AND co.nombre IN ('Azul', 'Negro'));

1-- 0) Mostrar el nombre de modelo completo (nombre fabricante, marca y modelo) de los 
--productos que han sido comprados en color verde por clientes de Madrid cuyo 
--nombre empieza por J y su primer apellido tiene una 'e':

SELECT f.nombre || ' ' || p.marca || ' ' || p.modelo AS modelo_completo
FROM producto p
JOIN fabricante f ON p.fabricante_codfabr = f.codfabr
JOIN pedido pe ON p.codprod = pe.producto_codprod
JOIN cliente c ON pe.cliente_dni = c.dni
JOIN color co ON pe.color_codcolor = co.codcolor
WHERE co.nombre = 'Verde'
AND c.codciudad = (Select codciudad from CIUDAD where nombre='Madrid')
AND c.nombre LIKE 'J%'
AND c.apellido1 LIKE '%e%';
