FROM DEPARTAMENTOS , EMPLEADOS;

/*Si nos fijamos en las columnas, vemos que nos salen 
todas las de la tabla departamento y todas las de la tabla empleados.--> LOS DATOS SE ESTAN MULTIPLICANDO
Vemos todas las posibles relaciones que podrian ser con los datos existentes. ESTO YA SE CONSIDERA = PRODUCTO
(estos resultados destan DESFORMALIZADOS, id departamento que es la clave foranea, que relaciona las 2 tablas
no se ha especificado que son lo mismo.)*/

--__--__--__--__--__--__--__--__--__--


--__--__--__--__--__--__--__--__--__--__-SELECCION/PROYECCION-__--__--__--__--__--

--Partamos de esta select

SELECT NOMBRE_DEPARTAMENTO
 FROM empleados, departamentos ;

/*venimos de la select anterior de producto, vemos 18 elementos,
todas las posibles relaciones existentes o no, con los datos iniciales
Vemos lo mismo que en la tabla anterior, 
pero solo la columna especifica -> NOMBRE_DEPARTAMENTO */

--"""WHERE(de la tabla empleados, ID_DEPARTAMENTO) SEA = (de la tabla departamentos, ID_DEPARTAMENTO)"""
/*aqui, añadiendo el where, descartamos otras opciones
y cuando comparamos ELEMENTOS IGUALES de DIFERENTES TABLAS estamos 
relacionando tablas -> ELEMENTO1 = ELEMENTO2 --> CLAVE FORANEA*/

SELECT NOMBRE_DEPARTAMENTO
FROM empleados, departamentos 
WHERE empleados.ID_DEPARTAMENTO = departamentos.ID_DEPARTAMENTO;
/*con eso seleccionamos la misma columna, pero asegurandonos la relacion entre tablas, 
porque hemos indicado la clave foranea que relacionaba nuestras 2 tablas*/

--esto es lo mismo, pero escribes menos jeje ;p :

SELECT NOMBRE_DEPARTAMENTO
FROM empleados e, departamentos d
WHERE e.ID_DEPARTAMENTO = d.ID_DEPARTAMENTO;

/*en resumen estamos DESCARTANDO INFO,

- La condicion que ponemos en el where es lo que relaciona las dos tablas
y dentro del where vemos la clave foranea.

- Como maximo tendremos los elementos de la tabla principal.

- En este caso la tabla departamentos complementa la info de la de empleados,
empleados es nuestra tabla principal, y recordemos que tenia 6 elementos.*/

SELECT *

-- veremos que nos da el mismo resultado que el select a explicar en este apartado
-->>>>>>>>>>>>>>>>>>>>> /////// 
--------------------------- select con JOIN --------
/*El JOIN nos sirve para referirnos en una sola linea 
a una relacion entre tablas, (claves foraneas) // Debemos tener en cuenta:
- En el apartado anterior, en el FROM pondremos todas las tablas a relacionar(en este caso trabajamos con 2 tablas)
Con el JOIN solo pondremos una tabla(la tabla que se relacionara con las escritas en los JOIN)
- Dentro del JOIN escribiremos la tabla a relacionar con la tabla inicial, 
(tabla inicial -> en el FROM--> en este ejemplo "EMPLEADOS")
- Dentro del JOIN, para referirnos a la relacion entre tablas ([elemento1 = elemento2] --> CLAVE FORANEA)
lo haremos con un ON ---> JOIN "tabla a relacionar" ON (CLAVE FORANEA)*/

SELECT *
FROM empleados 
JOIN departamentos ON empleados.ID_DEPARTAMENTO = departamentos.ID_DEPARTAMENTO;

/*Para verlo claro lo escribo asi, pero recoremos que asi escribimos menos ;d :*/

SELECT *
FROM empleados e
JOIN departamentos d ON e.ID_DEPARTAMENTO = d.ID_DEPARTAMENTO;

--__--__--__--__--__--__--__--__--__--



--__--__--__--__--__--__--__--__--__--__-JOIN(JOIN-WHERE) JOIN/SELECCION-__--__--__--__--__--

--Añadiremos un where a la query anterior
/*redordemos que añadiendo el where, descartamos otras opciones
- Sin el where, lo que se pide ver es una tabla con todos los datos
de las 2 tablas (empleado y departamento), pero teniendo en cuenta 
la relacion entre las tablas --> e.ID_DEPARTAMENTO = d.ID_DEPARTAMENTO -> Clave Foranea
- Ahora queremos ver especificamente los datos en los que coincida
que elID_DEPARTAMENTO, que corresponde a la table de empleados sea Mayor que 2 
(da igual si te refieres al ID ID_DEPARTAMENTO de empleados o departamento; sus valores son iguales)*/

SELECT *
FROM empleados e
JOIN departamentos d ON e.ID_DEPARTAMENTO = d.ID_DEPARTAMENTO
WHERE e.ID_DEPARTAMENTO > '2' ;

--__--__--__--__--__--__--__--__--__--



--__--__--__--__--__--__--__--__--__-JOIN(JOIN-WHERE) JOIN/SELECCION/PROYECCION-__--__--__--__--__--

-- cambiemos el * por "NOMBRE_DEPARTAMENTO"

pero solo la columna especifica -> NOMBRE_DEPARTAMENTO */
